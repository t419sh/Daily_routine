      energyBtn.classList.remove('active-energy');
      modeBadge.textContent = 'حالت عادی (۱۰۰٪)';
      modeBadge.style.color = 'var(--text-secondary)';
    }

    const nowMins = getCurrentMinutes();
    listEl.innerHTML = '';

    scheduleData.forEach(item => {
      const startM = timeToMinutes(item.start);
      const endM = timeToMinutes(item.end);
      const isNow = nowMins >= startM && nowMins < endM;
      const isChecked = !!state.checked[item.id];

      const itemDiv = document.createElement('div');
      itemDiv.className = `task-item ${isNow ? 'is-now' : ''} ${isChecked ? 'completed' : ''}`;

      let subtasksHtml = '';
      if (item.subtasks && item.subtasks.length > 0) {
        subtasksHtml = `<div class="subtasks">` + item.subtasks.map(sub => {
          const subDone = !!state.checked[sub.id];
          return `
            <div class="subtask-row ${subDone ? 'done' : ''}" onclick="toggleCheck('${sub.id}')">
              <div class="sub-check"></div>
              <span>${sub.text}</span>
            </div>
          `;
        }).join('') + `</div>`;
      }

      itemDiv.innerHTML = `
        <div class="task-main">
          <div class="checkbox-custom ${isChecked ? 'checked' : ''}" onclick="toggleCheck('${item.id}')"></div>
          <div class="task-content">
            <div class="task-top">
              <span class="time-badge">${state.energy80 ? item.timeLabel80 : item.timeLabel}</span>
              ${isNow ? '<span class="now-badge">در حال انجام</span>' : ''}
            </div>
            <div class="task-title">${item.title}</div>
            <div class="task-desc">${state.energy80 ? item.desc80 : item.desc}</div>
          </div>
        </div>
        ${subtasksHtml}
      `;

      listEl.appendChild(itemDiv);
    });

    updateProgress();
  }

  function toggleCheck(id) {
    haptic('light');
    state.checked[id] = !state.checked[id];
    saveState();
    renderSchedule();
  }

  function updateProgress() {
    let total = 0;
    let done = 0;

    scheduleData.forEach(item => {
      total++;
      if (state.checked[item.id]) done++;
      if (item.subtasks) {
        item.subtasks.forEach(sub => {
          total++;
          if (state.checked[sub.id]) done++;
        });
      }
    });

    const pct = total === 0 ? 0 : Math.round((done / total) * 100);
    document.getElementById('progressBar').style.width = pct + '%';
    document.getElementById('progressText').textContent = `${pct}٪ (${done} از ${total})`;
  }

  function resetAllTasks() {
    haptic('medium');
    state.checked = {};
    saveState();
    renderSchedule();
  }

  // ساعت و تاریخ زنده
  function updateClock() {
    const now = new Date();
    const timeStr = now.toLocaleTimeString('fa-IR', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    const dateStr = now.toLocaleDateString('fa-IR', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
    document.getElementById('liveClock').textContent = timeStr;
    document.getElementById('liveDate').textContent = dateStr;
  }

  // یادداشت سریع
  const notesEl = document.getElementById('quickNotes');
  notesEl.value = state.notes || '';
  notesEl.addEventListener('input', (e) => {
    state.notes = e.target.value;
    saveState();
  });

  // تایمر پومودورو و بوق ملایم
  let timerSeconds = 45 * 60;
  let defaultSeconds = 45 * 60;
  let timerInterval = null;

  function playBeep() {
    try {
      const ctx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(660, ctx.currentTime);
      gain.gain.setValueAtTime(0.2, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.8);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start();
      osc.stop(ctx.currentTime + 0.8);
    } catch (e) {}
  }

  function updateTimerDisplay() {
    const mins = Math.floor(timerSeconds / 60);
    const secs = timerSeconds % 60;
    document.getElementById('timerDisplay').textContent =
      `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
  }

  function setTimer(minutes) {
    haptic('light');
    clearInterval(timerInterval);
    timerInterval = null;
    document.getElementById('startStopBtn').textContent = 'شروع';
    defaultSeconds = minutes * 60;
    timerSeconds = defaultSeconds;
    updateTimerDisplay();
  }

  function toggleTimer() {
    haptic('medium');
    const btn = document.getElementById('startStopBtn');
    if (timerInterval) {
      clearInterval(timerInterval);
      timerInterval = null;
      btn.textContent = 'ادامه';
    } else {
      btn.textContent = 'توقف';
      timerInterval = setInterval(() => {
        if (timerSeconds > 0) {
          timerSeconds--;
          updateTimerDisplay();
        } else {
          clearInterval(timerInterval);
          timerInterval = null;
          btn.textContent = 'شروع';
          playBeep();
          haptic('heavy');
        }
      }, 1000);
    }
  }

  function resetTimer() {
    haptic('light');
    clearInterval(timerInterval);
    timerInterval = null;
    document.getElementById('startStopBtn').textContent = 'شروع';
    timerSeconds = defaultSeconds;
    updateTimerDisplay();
  }

  // اجرای اولیه
  updateClock();
  renderSchedule();
  setInterval(updateClock, 1000);
  setInterval(renderSchedule, 60000); // بروزرسانی نشانگر بازه فعال هر دقیقه
</script>
